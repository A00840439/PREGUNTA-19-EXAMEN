# Part b: Unit root tests (ADF)
def adf_test(series, title=''):
    result = adfuller(series, autolag='AIC')
    print(f'ADF Test for {title}:')
    print(f'ADF Statistic: {result[0]:.4f}')
    print(f'p-value: {result[1]:.4f}')
    print(f'Critical Values: {result[4]}')
    if result[1] < 0.05:
        print(f'{title} is stationary (reject null hypothesis)\n')
    else:
        print(f'{title} is non-stationary (fail to reject null hypothesis)\n')

# Test GS3M and GS6M in levels
adf_test(df['GS3M'], 'GS3M (Levels)')
adf_test(df['GS6M'], 'GS6M (Levels)')

# Test first differences
adf_test(df['GS3M'].diff().dropna(), 'GS3M (First Differences)')
adf_test(df['GS6M'].diff().dropna(), 'GS6M (First Differences)')

# Part c: Cointegration test (Engle-Granger)
# Step 1: Run OLS regression of GS6M on GS3M
from statsmodels.regression.linear_model import OLS
model = OLS(df['GS6M'], df['GS3M']).fit()
residuals = model.resid

# Step 2: ADF test on residuals
print('Cointegration Test (Engle-Granger):')
adf_test(residuals, 'Residuals from GS6M ~ GS3M')
# Note: Use Engle-Granger critical values (approx. -3.37 at 5% for two I(1) series)
